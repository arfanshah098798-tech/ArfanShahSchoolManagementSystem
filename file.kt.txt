class MainActivity : ComponentActivity() {
    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        setContent {
            MaterialTheme {
                Surface(
                    modifier = Modifier.fillMaxSize()
                ) {
                    SchoolApp()
                }
            }
        }
    }
}

enum class UserRole {
    CHAIRMAN,
    TEACHER,
    PARENT
}

data class Student(
    val name: String,
    val fatherName: String,
    val className: String,
    val rollNumber: String
)

@Composable
fun SchoolApp() {
    var currentRole by remember { mutableStateOf<UserRole?>(null) }
    var languageUrdu by remember { mutableStateOf(false) }

    if (currentRole == null) {
        LoginScreen(
            languageUrdu = languageUrdu,
            onLanguageChange = { languageUrdu = !languageUrdu },
            onLogin = { role -> currentRole = role }
        )
    } else {
        DashboardScreen(
            role = currentRole!!,
            languageUrdu = languageUrdu,
            onLogout = { currentRole = null }
        )
    }
}

@Composable
fun LoginScreen(
    languageUrdu: Boolean,
    onLanguageChange: () -> Unit,
    onLogin: (UserRole) -> Unit
) {
    var email by remember { mutableStateOf("") }
    var password by remember { mutableStateOf("") }

    Column(
        modifier = Modifier
            .fillMaxSize()
            .padding(24.dp),
        horizontalAlignment = Alignment.CenterHorizontally,
        verticalArrangement = Arrangement.Center
    ) {
        Text(
            text = "📘 ⭐",
            fontSize = 42.sp
        )

        Spacer(modifier = Modifier.height(12.dp))

        Text(
            text = if (languageUrdu)
                "ارفان شاہ اسکول مینجمنٹ سسٹم"
            else
                "Arfan Shah School Management System",
            fontSize = 21.sp,
            fontWeight = FontWeight.Bold
        )

        Spacer(modifier = Modifier.height(8.dp))

        Text(
            text = if (languageUrdu) "Demo Star School" else "Demo Star School",
            color = MaterialTheme.colorScheme.primary
        )

        Spacer(modifier = Modifier.height(24.dp))

        OutlinedButton(onClick = onLanguageChange) {
            Text(if (languageUrdu) "English" else "اردو")
        }

        Spacer(modifier = Modifier.height(18.dp))

        OutlinedTextField(
            value = email,
            onValueChange = { email = it },
            label = {
                Text(if (languageUrdu) "موبائل یا ای میل" else "Mobile or Email")
            },
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(12.dp))

        OutlinedTextField(
            value = password,
            onValueChange = { password = it },
            label = {
                Text(if (languageUrdu) "پاس ورڈ" else "Password")
            },
            modifier = Modifier.fillMaxWidth()
        )

        Spacer(modifier = Modifier.height(20.dp))

        Text(
            text = if (languageUrdu)
                "Demo login کے لیے نیچے کوئی بھی بٹن دبائیں"
            else
                "For demo login, select any role below",
            fontSize = 13.sp
        )

        Spacer(modifier = Modifier.height(12.dp))

        Button(
            onClick = { onLogin(UserRole.CHAIRMAN) },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (languageUrdu) "چیئرمین لاگ اِن" else "Chairman Login")
        }

        Spacer(modifier = Modifier.height(8.dp))

        Button(
            onClick = { onLogin(UserRole.TEACHER) },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (languageUrdu) "ٹیچر لاگ اِن" else "Teacher Login")
        }

        Spacer(modifier = Modifier.height(8.dp))

        Button(
            onClick = { onLogin(UserRole.PARENT) },
            modifier = Modifier.fillMaxWidth()
        ) {
            Text(if (languageUrdu) "والدین لاگ اِن" else "Parent Login")
        }
    }
}

@Composable
fun DashboardCard(title: String, value: String) {
    Card(
        modifier = Modifier
            .fillMaxWidth()
            .padding(vertical = 5.dp)
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(16.dp),
            horizontalArrangement = Arrangement.SpaceBetween
        ) {
            Text(
                text = title,
                fontWeight = FontWeight.Bold
            )

            Text(text = value)
        }
    }
}
